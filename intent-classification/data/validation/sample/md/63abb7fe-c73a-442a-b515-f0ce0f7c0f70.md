```python
class EnvVarSecretManager(SecretManager):
    """
    Secret manager that reads from environment variables.
    
    For Railway/PaaS deployments:
    - Detects USE_ENV_SECRETS=true
    - Returns plaintext secrets directly (Railway encrypts at platform level)
    - Skips AWS KMS decryption
    
    For AWS deployments:
    - Attempts KMS decryption if AWS credentials are available
    """
    
    async def get_secret(self, secret_name: str) -> Optional[str]:
        value = os.getenv(secret_name)
        if not value:
            return None
        
        # Railway/PaaS mode: return plaintext directly
        if os.getenv("USE_ENV_SECRETS", "").lower() == "true":
            logger.info(f"EnvVarSecretManager: Using Railway/PaaS mode for {secret_name}")
            return value
        
        # AWS mode: attempt KMS decryption
        if secret_name == "AUDIT_CRYPTO_SOFTWARE_KEY_MASTER_ENCRYPTION_KEY_B64":
            try:
                # Check if AWS credentials are available
                if not (os.getenv("AWS_ACCESS_KEY_ID") or os.getenv("AWS_REGION")):
                    logger.info(f"No AWS credentials found, treating {secret_name} as plaintext")
                    return value
                
                # Attempt KMS decryption
                import boto3
                kms = boto3.client('kms', region_name=os.getenv("AWS_REGION", "us-east-1"))
                response = kms.decrypt(CiphertextBlob=base64.b64decode(value))
                return base64.b64encode(response['Plaintext']).decode()
            except Exception as e:
                logger.warning(f"KMS decryption failed for {secret_name}, using as plaintext: {e}")
                return value
        
        return value
```
