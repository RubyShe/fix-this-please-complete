### Issue: Insufficient Input Validation
```javascript
// Bad
app.post('/user', (req, res) => {
    const { username } = req.body;
    db.query(`INSERT INTO users (username) VALUES ('${username}')`);
});

// Good
const Joi = require('joi');
app.post('/user', (req, res) => {
    const schema = Joi.object({
        username: Joi.string().alphanum().min(3).max(30).required()
    });
    const { error, value } = schema.validate(req.body);
    if (error) return res.status(400).send(error.details[0].message);
    db.query('INSERT INTO users (username) VALUES (?)', [value.username]);
});
```

### Issue: Insecure Direct Object Reference
```java
// Bad
@GetMapping("/document/{id}")
public Document getDocument(@PathVariable Long id) {
    return documentRepository.findById(id).orElseThrow();
}

// Good
@GetMapping("/document/{id}")
@PreAuthorize("hasPermission(#id, 'Document', 'read')")
public Document getDocument(@PathVariable Long id, Authentication auth) {
    Document doc = documentRepository.findById(id).orElseThrow();
    if (!doc.getOwnerId().equals(auth.getUserId())) {
        throw new AccessDeniedException("Not authorized");
    }
    return doc;
}
```
